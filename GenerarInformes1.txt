public int GenerarInformes(string pSesion, string pLoginNOSEUSA, string pIp, string vFECHA_DISCRIMINANTES, bool vBorrarTbenAjustes, string pIdioma, Dictionary<string, string> pParametrosCore, bool vieneDeRecalculoEnPlanificador = false)
        {
            #region AppendOracle
            string gAppendParaOracle;
            string comentarioError = "";
            if (Configuracion.getCache("AppendOracle") == "0") //AppendOracle
                gAppendParaOracle = "";
            else
                gAppendParaOracle = ConstantesBD.gAppendParaOracle; //POR DEFECTO
            #endregion

            bool vReintentarDesbloqueo = false;
            int vDuranteCuantoTiempoEnSegundos = 7200;
            int vCadaCuantoTiempoEnSegundos = 60;

            string gHintOracleContable = "";

            if (this.gbbdd == ConstantesBD.BD_Oracle)
            {
                gHintOracleContable = "" + Configuracion.getCache("hintContabilidadOracle");
                if (gHintOracleContable == "")
                {
                    gHintOracleContable = ConstantesBD.gHINTOracleConta;
                }
            }
            int contador_numero_errores = 0;
            int posActual = -1;
            ArrayList SubInformes;
            int nFilaTotal = 0;
            int nFilaConVariacionesAnteriores = 0;
            int bbdd = this.gbbdd;
            bool yaSeLlamoADesmarcar = false; //cuando se llame a desmarcar se pone a true para que si falla, no vuelva a llamar a desmarcar de nuevo en el cath (si hubiera un finally no sería necesario)

            //Inicializamos las cachés de casillas
            try
            {
                posActual = -2;
                if (gcolaInformes == null)
                    return 0;

                //Comprobamos que ninguno de los informes esté generándose
                //ADEMÁS RELLENA DATOS DEL INFORME!!!!
                if (!esPosibleGeneracion())
                    return -1;

                TraceGeneracionEstados.Show(this.gidSesion, this.gusuario, "Generacion Informes-INICIO: " + milisegundos(System.DateTime.Now), 2);

                int log_en_paralelo;
                if (hayAlgunaGeneracion(this.giM, this.gEsquema) == true)
                    log_en_paralelo = 1;
                else
                    log_en_paralelo = 0;

                #region leemos parametrizacion de reintento de desmarcar
                ParametrosConfigBBDD svcConfig = new ParametrosConfigBBDD();
                Dictionary<string, string> param_generales_configuracion = pParametrosCore;

                bool pPonerNulosEnCasillasVacias;
                if (param_generales_configuracion.ContainsKey("APP.GEN.VACIAS_A_NULOS"))
                {
                    string valorParam = param_generales_configuracion["APP.GEN.VACIAS_A_NULOS"];
                    if (valorParam=="1")
                        pPonerNulosEnCasillasVacias = true;
                    else
                        pPonerNulosEnCasillasVacias = false;
                }
                else
                {
                    pPonerNulosEnCasillasVacias = false;
                }

                if (param_generales_configuracion.ContainsKey("APP.GEN.DESBLOQUEO.REINTENTAR"))
                {
                    int valorParametroInt;
                    if (int.TryParse(param_generales_configuracion["APP.GEN.DESBLOQUEO.REINTENTAR"],out valorParametroInt))
                    {
                        vReintentarDesbloqueo = valorParametroInt == 1 ? true : false;
                    }
                    else
                    {
                        //el valor del parámetro no es un numero
                        vReintentarDesbloqueo = false;
                    }
                    
                }
                else
                {
                    vReintentarDesbloqueo = false;
                }

                if (param_generales_configuracion.ContainsKey("APP.GEN.DESBLOQUEO.REINTENTAR.TIEMPO_MAX_SEG"))
                {
                    int valorParametroInt;
                    if (int.TryParse(param_generales_configuracion["APP.GEN.DESBLOQUEO.REINTENTAR.TIEMPO_MAX_SEG"], out valorParametroInt))
                    {
                        vDuranteCuantoTiempoEnSegundos = valorParametroInt;
                    }
                    else
                    {
                        //el paraámetro no tiene un número
                        vDuranteCuantoTiempoEnSegundos = 7200;
                    }
                        
                }
                else
                {
                    vDuranteCuantoTiempoEnSegundos = 7200;
                }



                if (param_generales_configuracion.ContainsKey("APP.GEN.DESBLOQUEO.REINTENTAR.INTERVALO_SEG"))
                {
                    int valorParametroInt;
                    if (int.TryParse(param_generales_configuracion["APP.GEN.DESBLOQUEO.REINTENTAR.INTERVALO_SEG"], out valorParametroInt))
                    {
                        vCadaCuantoTiempoEnSegundos = valorParametroInt;
                    }
                    else
                    {
                        //el paraámetro no tiene un número
                        vCadaCuantoTiempoEnSegundos = 60;
                    }
                }
                else
                {
                    vCadaCuantoTiempoEnSegundos = 60;
                }

                int activarTraza;

                if (param_generales_configuracion.ContainsKey("APP.GEN.METODO.BLOQUES.ACTIVAR_TRAZA"))
                {
                    int valorParametroInt;
                    if (int.TryParse(param_generales_configuracion["APP.GEN.METODO.BLOQUES.ACTIVAR_TRAZA"], out valorParametroInt))
                    {
                        activarTraza = valorParametroInt;
                    }
                    else
                    {
                        //el paraámetro no tiene un número
                        activarTraza = 0;
                    }
                }
                else
                {
                    activarTraza = 0;
                }




                #endregion

                //#endif
                //marcamos los INFORMES a generar (quedarían bloqueados)
                //  ... y tambíen rellena IND_RECALCULO_DESPUES_USP, METODO_ALTERNATIVO, MetodoCalculo etc. y añade a la listaIndicesInformesARecalcular si IND_RECALCULO_DESPUES_USP = 1
                //  ....  pero sólo los rellena si no es un recálculo de totales (IND_SoloCalculadas =false)
                //  ... y por qué no se hace en la llamada anterior a esPosibleGeneracion() (allí se rellena MetodoCalculo si el estado lo tiene relleno, aquí sin embargo se rellena siempre)
                marcarGeneracion();

                //cachés de casillas (común para todos los informes!! para que si una casilla de un informe dependende de otra de otro informe que por alguna razón no se pudo generar, tampoco se pueda generar)
                ArrayList cacheCasillasPendientes = new ArrayList(); //casilla.ID_SUBINFORME.Trim() + "_" + casilla.ID_CASILLA.Trim();
                ArrayList cacheCasillasCalculadasPendientes = new ArrayList();
                ArrayList cacheCasillasCalculadasPendientesSimple = new ArrayList();
                Hashtable cacheSecuenciasInformesGenerados = new Hashtable();
                string idOrigen = "";
                string idOrigen2 = "";
                bool conVariaciones = false;
                posActual = -3;

                int indiceInformeGenerandose = -1;

                //Variables para almacenar tiempos en el caso de un Informe que tiene marcado volver a Recalcular casillas calculadas tras su generación (normalmente porque ejecute un procedimiento almacenado tras su generación, como el A7)
                long TiempoInicialDelPrimero = 0;
                long TiempoFinalPreparacionDelPrimero = 0;
                long TiempoFinalGeneracionDelPrimero = 0;
                long TiempoFinalDelPrimero = 0;

                if (activarLog)
                {
                    LogFichero.Write(ClsLog.LogLevel.Info, "Inicio Algoritmo de generación de estados", nivel);
                }

                //Algoritmo de generación de estados
                foreach (Informe informe in gcolaInformes)
                {
                    long TiempoInicial = milisegundos(System.DateTime.Now);
                    //Declaraciones de variables del método.
                    string strQuery = "";
                    string situacion = "";
                    int flag_volver_generar = 0;//sin errores
                    int log_tiempo_generacion = 0;
                    int log_tiempo_preparacion = 0;
                    int log_tiempo_fin = 0;
                    int log_recalculo = 0;
                    int log_cancelado = 0;
                    string log_opciones_config = "";
                    int log_n_errores = 0;
                    int log_n_registros = 0;
                    int log_n_querys = 0;
                    int contador_errores_informe_actual = 0;
                    long TiempoTotal = 0;
                    long TiempoFinal = 0;
                    bool generacionDeEsteInformeNoCancelada = false;

                    if (activarLog)
                    {
                        LogFichero.Write(ClsLog.LogLevel.Info, informe.ID_INFORME, nivel);
                    }

                    indiceInformeGenerandose++;

                    //Si SecActiva==0 : Informe que no se quiere generar, se ha marcado con 0 para bloquearlo
                    if (informe.IND_NoGenerar == true)
                        continue;

                    if ((generacionDeEsteInformeNoCancelada = generacionSigueEnStatusGeneracion(informe)) == false)
                        continue;

                    //Ahora sacamos los datos de codigo_entida y tipo_entidad de bbdd en lugar del fichero de configuración
                    //TODO PTE mejora: esto debería estar cacheado por si generamos N estados del mismo grupo""
                    //además, como se indica en getGruposMultientidad, debería pasarse la fecha para obtener los atributos del grupo a la fecha que interese en esta generación
                    Multientidad.GrupoMultientidad[] grupos = getGruposMultientidad(informe.IdGrupoEntidad);
                    string gCodigo_Entidad = grupos[0].CodigoEntidadExportacion; //  "" + Configuracion.getCache("codigoEntidad");
                    string gTipo_Entidad = grupos[0].TipoGrupo; //  "" + Configuracion.getCache("tipoEntidad");
                    string gLimiteInfAlfa = grupos[0].LimiteInferiorAlfa;
                    string gGrupoConsolidacionRRC = grupos[0].GrupoConsolidacionRRC;
                    bool gAgruparContabilidadEntidad = grupos[0].AgruparContabilidadEntidad;
                    bool gExcluirEntidadenEquivalencia = grupos[0].ExcluirEntidadenEquivalencia;

                    //esta variable es por grupo entidad, pero el buble es por informe, no por entidad, así que la calculamos en cada informe
                    int? metodoContratos = GetMetodoContratos(bbdd, gEsquema, informe.IdGrupoEntidad);

                    string _filtrosEspeciales = getFiltrosEspeciales(informe.ID_INFORME.Substring(0, 4), informe.FECHA_DATOS);

                    bool huboErroresEnLaPreparacion = false;

                    Hashtable cacheDependenciasOrigenes = new Hashtable(); //casilla.ID_SUBINFORME.Trim() + "_" + casilla.ID_CASILLA.Trim();
                    //bool generacionDeEsteInformeNoCancelada = false;
                    ReturnBorraGeneracionAnteriorYObtenerAjustes resultadoBorraGeneracionAnteriorYObtenerAjustes;
                    DataSet dsAjustesAnteriores = null;
                    string indicadorCargaEstadoCompleto;
                    //int contador_errores_informe_actual = 0;

                    //Para la nueva funcionalidad de excluir del recálculo las casillas que dependean de otros informes.
                    //en esta funcionalidad se quieren excluir los calculados a casillas ficticias del mismo estado
                    ArrayList cacheCasillasFicticiasParaExcluirDelRecalculo;

                    if (informe.IgnorarDependenciasConOtrosEstados)
                    {
                        cacheCasillasFicticiasParaExcluirDelRecalculo = new ArrayList();
                    }
                    else
                    {
                        cacheCasillasFicticiasParaExcluirDelRecalculo = null;
                    }



                    //log_n_registros = 0;
                    //log_n_querys = 0;

                    if (activarLog)
                    {
                        LogFichero.Write(ClsLog.LogLevel.Info, "Inicio Preparacion datos del informe", nivel);
                    }
                    #region Preparacion DATOS del INFORME
                    
                    if (informe.EsUnRecalculoTrasGeneracion == false)
                        TiempoInicialDelPrimero = TiempoInicial;

                    #region Auditoria
                    //PHP: Auditoria generación de un informe                    

                    string idGrupoEntidadAuditar = informe.IdGrupoEntidad.ToString();
                    string borrarAjustesAuditar = vBorrarTbenAjustes ? "True" : "False";
                    string vieneDeRecalculoEnPlanificadorAuditar = vieneDeRecalculoEnPlanificador ? "True" : "False";
                    
                    List<string> datosJson = new List<string>() { 
                        UtilsString.FormateaValorJson<string>("Informe", informe.ID_INFORME),
                        UtilsString.FormateaValorJson<string>("FechaDatos", informe.FECHA_DATOS),
                        UtilsString.FormateaValorJson<string>("Entidad", informe.IdGrupoEntidad),
                        UtilsString.FormateaValorJson<string>("BorrarAjustes", vBorrarTbenAjustes),
                        UtilsString.FormateaValorJson<string>("VieneDeRecalculoEnPlanificador", vieneDeRecalculoEnPlanificador)
                    };

                    List<string> datos = new List<string>() { informe.ID_INFORME, informe.FECHA_DATOS, idGrupoEntidadAuditar, borrarAjustesAuditar, vieneDeRecalculoEnPlanificadorAuditar };

                    TraceGeneracionEstados.AuditarESTATICAClone(pSesion, gusuario, pLoginNOSEUSA, pIdioma, pIp, ConstantesAuditoria.GENERACION_EXE_GENERAR, datosJson, datos);
                    #endregion

                    TraceGeneracionEstados.Show(this.gidSesion, this.gusuario, "Generacion INICIO DE " + informe.ID_INFORME + "(" + informe.NombreEntidad + ")," + informe.FECHA_DATOS + "," + informe.IdGrupoEntidad + ": " + TiempoInicial.ToString(), 4);

                    if (vFECHA_DISCRIMINANTES == "")
                        vFECHA_DISCRIMINANTES = informe.FECHA_DATOS;

                    //Establecemos la secuencia del informe (informe.SecActiva)
                    int misecARecalcular;

                    if (informe.EsUnRecalculoTrasGeneracion == true)
                    {
                        //En lugar de buscar la secuencia a generar USAMOS la MISMA que el informe anterior
                        Informe informeTmp = (Informe)this.gcolaInformes[indiceInformeGenerandose - 1];
                        informe.Dependencias = informeTmp.Dependencias;
                        informe.Impacto = informeTmp.Impacto;
                        informe.IND_NoGenerar = false;
                        informe.IND_RECALCULO_DESPUES_USP = "0";
                        informe.IND_SoloCalculadas = true;
                        informe.IND_TRATAMIENTO = informeTmp.IND_TRATAMIENTO;
                        informe.SecActiva = informeTmp.SecActiva;
                        informe.SECUENCIA = informeTmp.SECUENCIA;
                        informe.SITUACION = informeTmp.SITUACION;
                        informe.USP_ANTES_GENERAR = "";
                        informe.USP_DESPUES_GENERAR = "";//ya debe venir relleno informe.IdGrupoEntidad = informeTmp.IdGrupoEntidad;
                        informe.IgnorarDependenciasConOtrosEstados = false;
                    }
                    else
                    {

                        misecARecalcular = informe.SecActiva;

                        if (!vieneDeRecalculoEnPlanificador)
                        {
                            ObtenerSecuenciaActiva(informe);
                            if (informe.IND_SoloCalculadas == true && informe.SecActiva != misecARecalcular)
                                throw new Exception("Error. No es posible recalcular esta secuencia (Sólo se pueden recalcular secuencias Generadas-No Validadas");
                        }
                    }
                    posActual = -4;

                    #region Actualizamos STATUS GENERACION: SECUENCIA
                    int nReg;

                    if (informe.EsUnRecalculoTrasGeneracion == true)
                    {

                        nReg = this.giM.ExecuteNonQuery(Configuracion.getCache("cadenaConexion"), "UPDATE " + gEsquema +
                            "T_EBE_STATUS_GENERACION SET SEC_INFORME_GENERADO = " +
                            informe.SecActiva + ", TOTAL_NUM_CASILLAS = 0 WHERE ID_INFORME = '" +
                            informe.ID_INFORME + "' AND FECHA_DATOS = " +
                            ConstantesBD.DevuelveFechaEnFormatoBD(bbdd, "'" + informe.FECHA_DATOS + "'") +
                            " AND SEC_INFORME_GENERADO=" + informe.SecActiva + " AND ID_GRUPO_ENTIDAD=" + informe.IdGrupoEntidad.ToString());
                    }
                    else

                    {
                        //Marcamos el informe generándose con la secuencia y el número de casillas
                        //Si el nº total de casilla no es NULL significa que ha sido cancelado!!!
                        nReg = this.giM.ExecuteNonQuery(Configuracion.getCache("cadenaConexion"), "UPDATE " + gEsquema +
                            "T_EBE_STATUS_GENERACION SET SEC_INFORME_GENERADO = " +
                            informe.SecActiva + ", TOTAL_NUM_CASILLAS = 0 WHERE ID_INFORME = '" +
                            informe.ID_INFORME + "' AND FECHA_DATOS = " +
                            ConstantesBD.DevuelveFechaEnFormatoBD(bbdd, "'" + informe.FECHA_DATOS + "'") +
                            " AND SEC_INFORME_GENERADO IS NULL AND TOTAL_NUM_CASILLAS IS NULL AND ID_GRUPO_ENTIDAD=" + informe.IdGrupoEntidad.ToString());
                    }

                    if (nReg != 1)
                    {
                        //Cancelación o Registros erróneos!!!
                        TraceGeneracionEstados.Show(this.gidSesion, this.gusuario, "Cancelación del Informe:" + informe.ID_INFORME + "," + milisegundos(System.DateTime.Now), 3);
                        continue;
                    }
                    #endregion Actualizamos STATUS GENERACION: SECUENCIA

                    if (informe.EsUnRecalculoTrasGeneracion == false)
                    {
                        //Si es un recalculo tras una generación la secuencia usada ya estará informada en la caché, por lo que no la añadimos

                        cacheSecuenciasInformesGenerados.Add(informe.ID_INFORME + "_" + informe.FECHA_DATOS + "_" + informe.IdGrupoEntidad.ToString(), informe.SecActiva);
                    }


                    #region Obtener INFORMACION de Subinformes y Variaciones GLOBALES y LOCALES y DISCRIMINANTES VARIACIONES

                    //Obtenemos todos los subinformes para un informe dado
                    TraceGeneracionEstados.Show(this.gidSesion, this.gusuario, "Obteniendo Subinformes (USP_GET_SUBINFORMES " + informe.ID_INFORME + ")," + milisegundos(System.DateTime.Now), 5);
                    SubInformes = getSubInformes(informe.ID_INFORME);

                    //Por cada subinforme obtenemos sus variaciones globales
                    long[] totalNumCasillas = new long[SubInformes.Count];
                    int contador = 0;
                    Hashtable CacheTemporales = new Hashtable();  //Optimizaciones para el T.12: cacheado de tablas temporales
                    long TiempoFinalPreparacion = milisegundos(System.DateTime.Now);
                    posActual = -4;

                    //ReturnBorraGeneracionAnteriorYObtenerAjustes

                    resultadoBorraGeneracionAnteriorYObtenerAjustes = borraGeneracionAnteriorYObtenerAjustes(vBorrarTbenAjustes, SubInformes, informe.ID_INFORME, informe.SecActiva, informe.FECHA_DATOS, informe.IND_SoloCalculadas, informe.IdGrupoEntidad, informe.MetodoCalculo);
                    dsAjustesAnteriores = resultadoBorraGeneracionAnteriorYObtenerAjustes.dsAjustes;
                    indicadorCargaEstadoCompleto = resultadoBorraGeneracionAnteriorYObtenerAjustes.indicadorCargaEstadoCompleto;


                    TiempoFinalPreparacion = milisegundos(System.DateTime.Now);
                    try
                    {
                        if (informe.METODO_ALTERNATIVO)
                        {
                            MetricasGeneracion mg = GeneracionPHC_MetodoAlternativo(informe, vFECHA_DISCRIMINANTES);
                            log_n_registros += mg.nRegistros;
                            log_n_querys += mg.nConsultas;
                            contador_errores_informe_actual += mg.nErrores;
                            comentarioError = mg.MensajeError;

                            goto finInforme;
                        }

                        Hashtable cacheAliasVariaciones = new Hashtable();
                        DataSet dsTablasDiscComunes = null;
                        long totalNumCasillasTodosSubinformes = 0;

                        TraceGeneracionEstados.Show(this.gidSesion, this.gusuario, "Generacion ORIGENES de " + informe.ID_INFORME + "," + informe.FECHA_DATOS + "," + informe.IdGrupoEntidad + ": " + milisegundos(System.DateTime.Now), 4);

                        #region Ejecucion USP Antes de Generar
                        if (informe.IND_SoloCalculadas == false)
                        {
                            //Si es un recálculo no ejecutamos USPs Antes y Después
                            if (!String.IsNullOrEmpty(informe.USP_ANTES_GENERAR))
                            {
                                try
                                {
                                    TraceGeneracionEstados.Show(this.gidSesion, this.gusuario, "Ejecución USP Antes de Generar el informe " + informe.ID_INFORME + ": " + milisegundos(System.DateTime.Now), 4);

                                    //string gCodigo_Entidad = "" + Configuracion.getCache("codigoEntidad");
                                    //string gTipo_Entidad = "" + Configuracion.getCache("tipoEntidad");

                                    object[] parametrosUSPAntes = { informe.ID_INFORME, informe.FECHA_DATOS, vFECHA_DISCRIMINANTES, informe.SecActiva, informe.IdGrupoEntidad };
                                    string sentSQLUSPAntes;
                                    //14/02/2013: Permitir ejecutar pre y post procesos de fuera de un paquete
                                    //paquete.pl --> se ejecutará el pl en el paquete indicado a excepción del siguiente caso:
                                    //               si el paquete indicado es "T_EBE" (paquete de la aplicación), se sustituye por el paquete indicado en el configServidor (paqueteBD)
                                    //               por si la entidad ha renombrado el paquete de la aplicación (tanto en el configServidor como en BBDD)								
                                    //pl         --> se ejecutará el pl fuera de cualquier paquete
                                    if (informe.USP_ANTES_GENERAR.IndexOf(".") != -1
                                        && informe.USP_ANTES_GENERAR.Substring(0, informe.USP_ANTES_GENERAR.IndexOf(".")) == "T_EBE")
                                    {
                                        //Se sustituye el paquete "T_EBE" por el informado en el configServidor
                                        sentSQLUSPAntes = gEsquema + gPaquete + informe.USP_ANTES_GENERAR.Substring(informe.USP_ANTES_GENERAR.IndexOf(".") + 1, informe.USP_ANTES_GENERAR.Length - informe.USP_ANTES_GENERAR.IndexOf(".") - 1);
                                    }
                                    else
                                    {
                                        sentSQLUSPAntes = gEsquema + informe.USP_ANTES_GENERAR;
                                    }
                                    //								Anterior a la modificación del 14/02/2013
                                    //								if (informe.USP_ANTES_GENERAR.IndexOf(".")>0)
                                    //								{
                                    //									sentSQLUSPAntes = gEsquema + informe.USP_ANTES_GENERAR;
                                    //								}
                                    //								else
                                    //								{
                                    //									sentSQLUSPAntes = gEsquema + gPaquete + informe.USP_ANTES_GENERAR;
                                    //								}
                                    this.giM.ExecuteNonQuery(Configuracion.getCache("cadenaConexion"), sentSQLUSPAntes, parametrosUSPAntes);
                                }
                                catch (Exception ex)
                                {
                                    //throw new Exception("Error en la Ejecución del USP (" + informe.USP_ANTES_GENERAR +") Antes de Generar el informe " + informe.ID_INFORME,ex);
                                    huboErroresEnLaPreparacion = true;
                                    contador_errores_informe_actual++;
                                    TraceGeneracionEstados.Show(this.gidSesion, this.gusuario, "Error en generación(Procedimiento Antes :" + informe.USP_ANTES_GENERAR + ",Informe:" + informe.ID_INFORME + ",Esquema:" + gEsquema + ",Paquete:" + gPaquete + "):" + ex.Message, 1);
                                    comentarioError = "Error en generación(Procedimiento Antes :" + informe.USP_ANTES_GENERAR + ",Informe:" + informe.ID_INFORME + ",Esquema:" + gEsquema + ",Paquete:" + gPaquete + "):" + ex.Message;

                                    try
                                    {
                                        if (informe.IND_RECALCULO_DESPUES_USP == "1")
                                        {
                                            //En este caso, en la cola de informes, el SIGUIENTE ESTADO SIEMPRE es el ACTUAL pero en modo RECALCULO
                                            //--> Marcar el siguiente para que no me lo genere
                                            Informe tmpInf = (Informe)this.gcolaInformes[indiceInformeGenerandose + 1];
                                            tmpInf.IND_NoGenerar = true;
                                        }
                                    }
                                    catch (Exception ex2)
                                    {
                                        contador_errores_informe_actual++;
                                        TraceGeneracionEstados.Show(this.gidSesion, this.gusuario, "Error desmarcar el Recálculo tras fallas el USP Después del Informe:" + informe.USP_DESPUES_GENERAR + ",Informe:" + informe.ID_INFORME + "):" + ex2.Message, 1);
                                    }
                                }
                            }
                        }
                        #endregion

                        //PROPUESTA CAMBIO (JCG)--> No cachear alias ni queries de discriminantes: Rellenar directamente en este
                        //bucle los Filtros Globales y Locales
                        if (!informe.MetodoCalculo.Equals("1") &&
                            !(informe.IND_SoloCalculadas && informe.MetodoRecalculoTotales.Equals("1"))
                            )
                        {
                            foreach (SubInforme subInforme in SubInformes)
                            {
                                posActual = -5;
                                //Se añaden las variaciones globales a los subinformes
                                TraceGeneracionEstados.Show(this.gidSesion, this.gusuario, "Obteniendo Variaciones Subinforme (USP_GET_VARIACIONES_GLOBALES " + subInforme.ID_SUBINFORME + ")," + milisegundos(System.DateTime.Now), 5);
                                getVariacionesGlobales(cacheAliasVariaciones, subInforme);
